## 一、一个场景的骨架

```tsx
import React from "react";
import { interpolate, useCurrentFrame } from "remotion";
import { C, SERIF, Sfx, Shell, useLines, useWordAt } from "../kit";
import { useTalk } from "../cast/talk";
import { Emote, Sprite } from "../v16/Sprite";
import { TagDiv } from "../ep17/scenesA";
import { Card, FOOT, GOLD, Keyword, PaperTown, popIn, RED, Screen, Shot, Verdict } from "./world";

const CL = { extrapolateLeft: "clamp", extrapolateRight: "clamp" } as const;

// 场景说明：这一步回答什么问题；镜头 1 / 2 / 3 各演什么
export const STrade: React.FC = () => {
  const f = useCurrentFrame();
  const L = useLines("trade");               // 场景 id 和 script.json 一致
  const W = useWordAt("trade");
  const M = useTalk("trade");                // 角色张嘴
  const s = [0, 1, 2, 3].map((i) => L.s(i)); // 每句开始帧
  const wTen = W(0, "一缝就是十天"), wBuy = W(3, "换回多得多");  // 锚点词：这句里唯一
  return (
    <Shell dur={L.dur} enter="page" cam="still" zoom={0} kicks={[wBuy]}>
      <PaperTown f={f} />                                   {/* 冷调：<PaperTown f={f} dusk={0.26} /> */}
      {/* 镜头 1：案例 */}
      <Shot f={f} a={0} b={s[2]}>
        <Screen>{/* 本集道具：田、镇子、粮袋……（彩铅画法） */}</Screen>
        <Sprite id="folk_farmer" view="front" x={300} y={FOOT} h={600} seed={61} talk={M("农夫", f)} jolts={[wTen]} />
        <Emote x={370} y={FOOT - 640} kind="sweat" at={wTen} dur={40} />
        <TagDiv x={300} y={FOOT - 690} text="自己缝：十天" col={GOLD} k={popIn(f, wTen - 4)} size={34} />
        <Keyword text="十天 vs 当天" at={s[0]} out={s[2]} />       {/* ≤8 字，不复读字幕 */}
      </Shot>
      {/* 镜头 2：异议（小满单独入镜，放大） */}
      <Shot f={f} a={s[2]} b={s[3]} d={4}>
        <Sprite id="xiaoman3" view="tq_r" x={540} y={FOOT} h={620} seed={1} talk={M("小满", f)} />
        <Emote x={610} y={FOOT - 660} kind="?" at={s[2] + 4} />
      </Shot>
      {/* 镜头 3：解释换方向 + 结论章 */}
      <Shot f={f} a={s[3]} b={L.dur + 20}>
        <Card x={60} y={480} w={960} h={300} k={popIn(f, wBuy - 6)} accent={GOLD}>{/* 讲道理的图 */}</Card>
        <Verdict i={0} text="两边都赚" at={W(3, "换回多得多") + 30} />
      </Shot>
      <Sfx at={wTen} name="tick" v={0.26} />
      <Sfx at={wBuy} name="coin" v={0.3} />
    </Shell>
  );
};
```

### 要点
- **每个场景都以 `<PaperTown>` 打底**；一句话里的镜头用 `<Shot>` 切（前后叠化），镜头基本静止
- 元素在锚点前 4–10 帧出现（`popIn(f, w - 6)`），**不晚于旁白**；锚点词在这句里唯一
- 角色：只放相关的一两个、`h` 600 左右；说话用 `talk={M("名字", f)}`；吓一跳 `jolts`；头顶 `Emote`
- 每个镜头主道具之外至少还有一个在动的东西（人走进来、东西飞过去 `Fly`、货出现）
- 读数分段跳，不显示插值中间值
- 结论章 `Verdict`、公式卡 `FormulaXX` 在本集 `world.tsx` 里照上一集改

## 二、开头场景（hook）的写法要点（`样板\场景代码_ep23\scenesA.tsx` 的 `SHook`）
- `<Shell enter="none" cam="still" zoom={0}>`；第 0 帧就有画面
- 镜头 1（0 到第二句）：一行 HTML 大字（`fontFamily: SERIF`、76 号、`WebkitTextStroke: 5px C.paper`、`paintOrder: "stroke fill"`），念到关键词变 `RED`；下面一个彩铅物件（`CountryTown`）；`<Sfx at={2} name="ding" />`
- 镜头 2（第二句起，约 2.5 秒）：两个角色放大入场、头顶“！”、各举一个道具、`Keyword` 出现
- menu 场景：镜头 3 “《国富论》里说”+ 物件动 + `RedStamp`；镜头 4 三问卡（`Card` × 3，间隔 `Math.min(W(1,"…"), r1 + 45)` 保证 ≤1.5 秒，第三张红边放大）

## 三、Video.tsx
```tsx
import React from "react";
import { Episode } from "../Episode";
import type { Timeline } from "../kit";
import { Avatar16 } from "../v16/Sprite";
import timeline from "./timeline.json";
import { SHook, SMenu, STrade } from "./scenesA";
import { SGrow, SLand } from "./scenesB";
import { SCta, SEnding } from "./scenesC";
import { PartBar, PARTS } from "./world";

export const TL23 = timeline as Timeline;
const SCENES: Record<string, React.FC> = { hook: SHook, menu: SMenu, trade: STrade, grow: SGrow, land: SLand, ending: SEnding, cta: SCta };
const WHO: Record<string, string> = { 小满: "xiaoman3", 小雪: "female2", 农夫: "folk_farmer", 阿禾: "ahe" };  // 字幕名 → 角色图

const sc = (id: string) => TL23.scenes.find((s) => s.id === id)!;
const line = (id: string, i: number) => TL23.lines.filter((l) => l.scene === id)[i].from;
const lastLine = (id: string) => { const ls = TL23.lines.filter((l) => l.scene === id); return ls[ls.length - 1].from; };

const BAR = (
  <PartBar parts={PARTS} from={line("menu", 1)} to={sc("cta").from}
    ticks={[lastLine("trade"), lastLine("grow"), lastLine("land")]}
    spans={[[sc("trade").from, sc("grow").from], [sc("grow").from, sc("land").from], [sc("land").from, sc("ending").from]]} />
);

export const EP23: React.FC = () => (
  <Episode tl={TL23} scenes={SCENES} progressAt="none" mix="v3" cold look="paper" overlay={BAR}
    badge={(who) => (WHO[who] ? <Avatar16 id={WHO[who]} /> : null)} />
);
```
